Exercise 4: Seating capacity with a default value of 50
================================================================================

CODE
class Vehicle:
    def __init__(self, name, max_speed, mileage):
        self.name = name
        self.max_speed = max_speed
        self.mileage = mileage

    def seating_capacity(self, capacity):
        return f"The seating capacity of a {self.name} is {capacity} passengers"

class Bus(Vehicle):
    def seating_capacity(self, capacity=50):
        return super().seating_capacity(capacity)

bus = Bus("bus", 180, 12)
print(bus.seating_capacity())

OUTPUT
The seating capacity of a bus is 50 passengers

================================================================================
Exercise 6: Override fare() with a 10% maintenance charge
================================================================================

CODE
class Vehicle:
    def __init__(self, name, mileage, capacity):
        self.name = name
        self.mileage = mileage
        self.capacity = capacity

    def fare(self):
        return self.capacity * 100

class Bus(Vehicle):
    def fare(self):                         # override, calls the parent method
        return super().fare() * 1.10        # + 10% maintenance charge

School_bus = Bus("School Volvo", 12, 50)
print("Total Bus fare is:", School_bus.fare())

OUTPUT
Total Bus fare is: 5500.0

Exercise 8: Is School_bus also an instance of Vehicle?
================================================================================

CODE
class Vehicle:
    def __init__(self, name, mileage, capacity):
        self.name = name
        self.mileage = mileage
        self.capacity = capacity

class Bus(Vehicle):
    pass

School_bus = Bus("School Volvo", 12, 50)
print(isinstance(School_bus, Vehicle))

OUTPUT
True

(Yes - Bus inherits from Vehicle, so every Bus object is also a Vehicle
instance, which is why isinstance() returns True.)
